alias n2oSensor      d0   # Pipe Analyzer on the N2O buffer tank (Pressure)
alias filtrationN2O  d1   # Filtration unit, N2O filter - pulls N2O out of the Ice Crusher's gas output
alias iceCrusher     d2   # Ice Crusher, Nitrice-fed
alias n2oReliefValve d3   # Digital Valve, N2O buffer -> exterior, overpressure backstop

# N2O condenses far more easily than anything else piped in this build - its triple point
# sits around -21C (confirmed via the in-game Stationpedia phase graph), and the liquid
# region opens up fast above that as pressure climbs. Unlike the CO2 line, which is
# deliberately compressed and cooled to force it to condense, this branch has to stay LOW
# pressure and near room temperature on purpose, or it liquefies in the pipe instead of
# reaching the tank as usable gas. Keep this whole run (Ice Crusher, Filtration, buffer
# tank) indoors at room temperature - don't route it past anything cold like the exterior
# radiator run on the CO2 line.
#
# Buffer target kept well below every other supply-side buffer in this build on purpose -
# this is a placeholder, not a verified safe ceiling. Watch the buffer's actual phase
# state once gas is flowing (Stationpedia graph or a Pipe Analyzer reading) and lower
# N2O_TARGET further if it's still condensing at this pressure.
define N2O_TARGET 1000
define N2O_LOW    800

# Relief backstop, same pattern as every other tank in this build - trips well above the
# operating target, vents to exterior. Kept modest here on purpose: the point isn't tank
# burst protection (nowhere near the 60,000 kPa tank rating), it's catching a runaway
# condition before the buffer drifts further past the phase-safe operating zone above.
define N2O_RELIEF_HIGH 1500
define N2O_RELIEF_LOW  1300

main:
yield

l r0 n2oSensor Pressure
l r1 iceCrusher On
beqz r1 n2o_off
slt r2 r0 N2O_TARGET
j n2o_set
n2o_off:
slt r2 r0 N2O_LOW
n2o_set:
s iceCrusher On r2
s filtrationN2O On r2

l r3 n2oReliefValve On
beqz r3 nrv_off
sgt r4 r0 N2O_RELIEF_LOW
j nrv_set
nrv_off:
sgt r4 r0 N2O_RELIEF_HIGH
nrv_set:
s n2oReliefValve On r4

j main
