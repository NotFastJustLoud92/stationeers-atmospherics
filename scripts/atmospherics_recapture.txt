alias sensor              d0   # Gas Sensor, shared room
alias recyclePump         d1   # Volume Pump, pulls room air into the filtration manifold
alias filtrationCO2       d2   # Filtration unit, CO2 filter only - recovered CO2 -> Area 400 TK-401
alias filtrationPollutant d3   # Filtration unit, Pollutant filter - captured gas vented to exterior
alias filtrationVolatiles d4   # Filtration unit, Volatiles filter - captured gas vented to exterior

# Every gas this loop touches goes to a tank now, not back into the room - the only thing
# that ever adds gas to the room atmosphere is Regulation's three pumps (atmospherics_
# regulation.ic10). That's why there's no "remainder" path here anymore: whatever this pump
# pulls in gets fully sorted - O2, CO2, and N2 each to their own tank, Pollutant and
# Volatiles vented to exterior.
#
# O2 and N2 are controlled by a companion housing (atmospherics_recapture_riders.ic10), not
# this one - one IC housing only has 6 pins and this loop needs 7 devices (sensor, pump, and
# 5 filters), so it's split in two. The rider housing just mirrors this housing's recyclePump
# On state onto the O2 and N2 filters; all the real decision logic lives here.
#
# CO2 keeps its hysteresis band, same as before - CO2 belongs in the room mix at 20%, only
# the excess above target gets skimmed. Pollutant and Volatiles have no target at all though
# - there's no "acceptable" amount of either, so those two just trigger on bare presence
# (ratio > 0), no hysteresis band to hover near.
define CO2_HIGH 20
define CO2_LOW  19

main:
yield

l r0 sensor Pressure
l r1 sensor RatioCarbonDioxide
mul r1 r1 r0   # CO2 partial pressure

l r2 filtrationCO2 On
beqz r2 co2_off
sgt r3 r1 CO2_LOW
j co2_set
co2_off:
sgt r3 r1 CO2_HIGH
co2_set:
s filtrationCO2 On r3

l r4 sensor RatioPollutant
sgt r5 r4 0
s filtrationPollutant On r5

l r6 sensor RatioVolatiles
sgt r7 r6 0
s filtrationVolatiles On r7

# Pump runs whenever any one of the three triggers above is active
max r8 r3 r5
max r8 r8 r7
s recyclePump On r8

j main
