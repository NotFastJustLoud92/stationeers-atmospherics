alias sensor_800              d0   # Gas Sensor, shared w/ GS-601 (atmospherics_regulation.ic10) via 2nd cable
alias recyclePump_800         d1   # Volume Pump, pulls room air into filtration manifold
alias filtrationCO2_800       d2   # CO2 filter -> Area 400 TK-401
alias filtrationPollutant_800 d3   # Pollutant filter -> Pollutant tank
alias filtrationMethaneH2_800 d4   # Methane+Hydrogen (2 cartridges, 1 unit) -> mix tank

# Everything captured goes to a tank, nothing returns to room - only Regulation adds gas.
# RatioVolatiles is now RatioMethane (game rename). Hydrogen is separate from Methane.
# O2/N2 filters live on atmospherics_recapture_riders.ic10 (_800r) - it independently
# gates each on its OWN excess partial pressure now, not on this housing's pump state.
# CO2/N2/O2 skim above target (hysteresis); Pollutant/Methane/H2 trigger on any presence.
# N2/O2 here (r12/r15) only feed the pump decision - the rider recomputes both from its
# own sensor cable to gate its own filters, so a CO2 flood doesn't drag N2/O2 down too.
define CO2_HIGH 20
define CO2_LOW  19
define N2_HIGH  60
define N2_LOW   58
define O2_HIGH  20
define O2_LOW   19

main:
yield

l r0 sensor_800 Pressure
l r1 sensor_800 RatioCarbonDioxide
mul r1 r1 r0

l r2 filtrationCO2_800 On
beqz r2 co2_off
sgt r3 r1 CO2_LOW
j co2_set
co2_off:
sgt r3 r1 CO2_HIGH
co2_set:
s filtrationCO2_800 On r3

l r4 sensor_800 RatioPollutant
sgt r5 r4 0
s filtrationPollutant_800 On r5

l r6 sensor_800 RatioMethane
sgt r7 r6 0
l r8 sensor_800 RatioHydrogen
sgt r9 r8 0
max r7 r7 r9
s filtrationMethaneH2_800 On r7

l r11 sensor_800 RatioNitrogen
mul r11 r11 r0
beqz r12 n2_off
sgt r13 r11 N2_LOW
j n2_set
n2_off:
sgt r13 r11 N2_HIGH
n2_set:
move r12 r13

l r14 sensor_800 RatioOxygen
mul r14 r14 r0
beqz r15 o2_off
sgt r13 r14 O2_LOW
j o2_set
o2_off:
sgt r13 r14 O2_HIGH
o2_set:
move r15 r13

max r10 r3 r5
max r10 r10 r7
max r10 r10 r12
max r10 r10 r15
s recyclePump_800 On r10

j main
