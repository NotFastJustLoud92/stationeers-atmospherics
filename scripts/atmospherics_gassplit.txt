alias n2Sensor_300      d0   # Pipe Analyzer on TK-302 (N2 tank)
alias filtrationN2_300  d1   # Filtration unit, N2 filter - splits N2 out of the post-CV-301...303 gas stream
alias o2Sensor_300      d2   # Pipe Analyzer on the O2 output line (shared network feeding Area 500's TK-501)
alias filtrationO2_300  d3   # Filtration unit, O2 filter - splits O2 out of what's left after the N2 filter
alias n2ReliefValve_300 d4   # Digital Valve, TK-302 -> exterior, overpressure backstop
alias o2ReliefValve_300 d5   # Digital Valve, O2 network -> exterior, overpressure backstop

# Area 300 used to dump N2 and O2 into one shared tank and let Regulation's two separate
# pumps both draw from that same mixed pipe - that doesn't actually separate anything, a
# pump just moves whatever's in the pipe, it can't selectively pull one gas out of a mix.
# Chained an N2 filter then an O2 filter off the post-condensation gas line instead, each
# into its own hysteresis-gated buffer/network, so Regulation's n2Pump and o2Pump each get
# a genuinely pure supply instead of both drawing from the same bottle.
# Placeholder buffer targets, same pattern as the other supply-side buffers - tune once running.
#
# Every alias in this build is suffixed by area number (_100, _200, etc.) - all IC housings
# share one network, so alias names need to be globally unique, not just unique within a
# single script. Area 500 also reads/writes TK-302 and the O2 network from its own housing
# (atmospherics_icecrusher.ic10) - that script's o2Sensor_500/n2Sensor_500 point at the same
# physical devices as this script's o2Sensor_300/n2Sensor_300, just with distinct alias names
# since they're two different scripts.
define N2_TARGET 40000
define N2_LOW    38000
define O2_TARGET 40000
define O2_LOW    38000

# Relief backstops, same pattern as every other tank in this build now - trip well above
# the normal operating target but with real margin under the 60,000 kPa tank burst rating.
# Vents to exterior, not the room.
define N2_RELIEF_HIGH 50000
define N2_RELIEF_LOW  48000
define O2_RELIEF_HIGH 50000
define O2_RELIEF_LOW  48000

main:
yield

l r0 n2Sensor_300 Pressure
l r1 filtrationN2_300 On
beqz r1 n2_off
slt r2 r0 N2_TARGET
j n2_set
n2_off:
slt r2 r0 N2_LOW
n2_set:
s filtrationN2_300 On r2

l r3 o2Sensor_300 Pressure
l r4 filtrationO2_300 On
beqz r4 o2_off
slt r5 r3 O2_TARGET
j o2_set
o2_off:
slt r5 r3 O2_LOW
o2_set:
s filtrationO2_300 On r5

l r6 n2ReliefValve_300 On
beqz r6 n2rv_off
sgt r7 r0 N2_RELIEF_LOW
j n2rv_set
n2rv_off:
sgt r7 r0 N2_RELIEF_HIGH
n2rv_set:
s n2ReliefValve_300 On r7

l r8 o2ReliefValve_300 On
beqz r8 o2rv_off
sgt r9 r3 O2_RELIEF_LOW
j o2rv_set
o2rv_off:
sgt r9 r3 O2_RELIEF_HIGH
o2rv_set:
s o2ReliefValve_300 On r9

j main
