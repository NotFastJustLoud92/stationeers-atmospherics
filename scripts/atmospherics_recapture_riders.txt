alias sensor_800r       d0   # Gas Sensor, SAME physical device as sensor_800/GS-601, via a 3rd cable
alias filtrationO2_800r d1   # O2 filter only - recovered O2 -> Area 500 O2 tank bank
alias filtrationN2_800r d2   # N2 filter only - recovered N2 -> Area 300 TK-302

# Each filter gates on its OWN gas's excess partial pressure now (same math
# atmospherics_recapture.ic10 uses for CO2/N2/O2 on the pump), instead of blindly
# mirroring the pump's On state. So a CO2 flooding event (e.g. 70% CO2/18% N2/12%
# O2, overpressure) only trips the CO2 filter - N2/O2 stay off since their own
# partial pressure isn't over target, even though the pump is running for CO2.
# If the ratio later settles near 60/20/20 while pressure stays elevated, all
# three gases' partial pressures cross their own threshold together, so all three
# filters naturally turn on together too - no separate "stable ratio" check
# needed, it falls out of using partial pressure (ratio x total pressure) rather
# than raw ratio. No pump read needed here anymore - atmospherics_recapture.ic10
# already brings the pump online whenever N2 or O2 is independently in excess,
# since it watches the same physical sensor with the same thresholds.
define N2_HIGH 60
define N2_LOW  58
define O2_HIGH 20
define O2_LOW  19

main:
yield

l r0 sensor_800r Pressure
l r1 sensor_800r RatioNitrogen
mul r1 r1 r0
l r2 filtrationN2_800r On
beqz r2 n2_off
sgt r3 r1 N2_LOW
j n2_set
n2_off:
sgt r3 r1 N2_HIGH
n2_set:
s filtrationN2_800r On r3

l r4 sensor_800r RatioOxygen
mul r4 r4 r0
l r5 filtrationO2_800r On
beqz r5 o2_off
sgt r6 r4 O2_LOW
j o2_set
o2_off:
sgt r6 r4 O2_HIGH
o2_set:
s filtrationO2_800r On r6

j main
