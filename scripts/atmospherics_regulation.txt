alias sensor  d0   # Gas Sensor, shared room
alias n2Pump  d1   # Turbo Volume Pump
alias o2Pump  d2   # Turbo Volume Pump
alias co2Pump d3   # Turbo Volume Pump

# CO2 target raised from 5% to 20% - Ryan confirmed 20% CO2 is fine for humans in this
# game, so it's now actively maintained at that level instead of the old, more
# conservative 5%. N2 absorbed the difference (75 -> 60); O2 unchanged.
#
# Pumps upgraded from Volume Pump to Turbo Volume Pump (2026-08-07) for faster response
# keeping up with the room's demand swings. No logic change - same On logic type as a
# regular Volume Pump, only the device placed and wired changes.
define N2_TARGET  60
define N2_LOW     58
define O2_TARGET  20
define O2_LOW     19
define CO2_TARGET 20
define CO2_LOW    19

main:
yield

l r0 sensor Pressure
l r1 sensor RatioNitrogen
l r2 sensor RatioOxygen
l r3 sensor RatioCarbonDioxide

mul r1 r1 r0   # N2 partial pressure
mul r2 r2 r0   # O2 partial pressure
mul r3 r3 r0   # CO2 partial pressure

l r4 n2Pump On
beqz r4 n2_off
slt r5 r1 N2_TARGET
j n2_set
n2_off:
slt r5 r1 N2_LOW
n2_set:
s n2Pump On r5

l r4 o2Pump On
beqz r4 o2_off
slt r5 r2 O2_TARGET
j o2_set
o2_off:
slt r5 r2 O2_LOW
o2_set:
s o2Pump On r5

l r4 co2Pump On
beqz r4 co2_off
slt r5 r3 CO2_TARGET
j co2_set
co2_off:
slt r5 r3 CO2_LOW
co2_set:
s co2Pump On r5

j main
