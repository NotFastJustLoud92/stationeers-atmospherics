alias tankPA_800mh2              d0   # Pipe Analyzer on inline buffer tank (Pressure)
alias pump_800mh2                d1   # Volume Pump, blue line -> inline tank
alias reliefValve_800mh2         d2   # Digital Valve, inline tank -> exterior, overpressure backstop
alias filtrationMethaneH2_800mh2 d3   # Filtration unit, BOTH cartridges - inline tank -> mix tank
alias sensor_800mh2              d4   # Gas Sensor, SAME physical device as GS-601, via its own cable

# One of 5 independent per-gas branches replacing the old single shared-pump recapture
# design (2026-08-09) - see atmospherics_recapture_co2.ic10 for the full reasoning,
# same pattern here. Pump keeps the inline tank pre-pressurized on its own hysteresis;
# filter gates independently on bare presence of Methane OR Hydrogen in the room (no
# target level - Volatile Ice safety contingency, any trace should be scrubbed), not
# on tank pressure. filtrationMethaneH2_800mh2 is ONE unit, both cartridges loaded.
# Placeholder tank/relief targets - no destination-tank pressure to reference yet
# (TK-803 has no active target), tune once running.
define TANK_TARGET 2000
define TANK_LOW    1800
define RELIEF_HIGH 3000
define RELIEF_LOW  2800

main:
yield

l r0 tankPA_800mh2 Pressure
l r1 pump_800mh2 On
beqz r1 pump_off
slt r2 r0 TANK_TARGET
j pump_set
pump_off:
slt r2 r0 TANK_LOW
pump_set:
s pump_800mh2 On r2

l r3 reliefValve_800mh2 On
beqz r3 relief_off
sgt r4 r0 RELIEF_LOW
j relief_set
relief_off:
sgt r4 r0 RELIEF_HIGH
relief_set:
s reliefValve_800mh2 On r4

l r5 sensor_800mh2 RatioMethane
sgt r6 r5 0
l r7 sensor_800mh2 RatioHydrogen
sgt r8 r7 0
max r6 r6 r8
s filtrationMethaneH2_800mh2 On r6

j main
